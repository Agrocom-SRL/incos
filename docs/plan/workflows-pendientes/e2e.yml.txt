name: E2E

# Smoke E2E del frontend (Playwright + Chromium) contra `vite preview` con la API
# interceptada (page.route): sin backend, sin BD, sin red. Complementa a ci.yml.
on:
  pull_request:
    branches: [master, develop, design]
    types: [opened, synchronize, reopened, ready_for_review]

permissions:
  contents: read

concurrency:
  group: e2e-${{ github.ref }}
  cancel-in-progress: true

jobs:
  e2e:
    name: Frontend (Playwright)
    runs-on: ubuntu-24.04
    timeout-minutes: 20
    defaults:
      run:
        working-directory: frontend
    steps:
      - uses: actions/checkout@v7
      - uses: actions/setup-node@v7
        with:
          node-version: 22
          cache: npm
          cache-dependency-path: frontend/package-lock.json
      - run: npm ci
      - name: Instalar Chromium
        run: npx playwright install --with-deps chromium
      # playwright.config.js construye y levanta `vite preview` (webServer).
      - name: Pruebas E2E
        run: npm run test:e2e
      - name: Reporte de Playwright
        if: failure()
        uses: actions/upload-artifact@v4
        with:
          name: playwright-report
          path: frontend/playwright-report/
          retention-days: 7
